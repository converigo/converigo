        }

        let message = `Failed to convert prompt file ${file2.path} to slash command: ${e14 instanceof Error ? e14.message : e14}`;

          const errorDetails = e14 instanceof Error ? e14.message : String(e14);

    } catch (e14) {
      localErrors.push({
        fatal: false,
        message: `Failed to load model:

${e14 instanceof Error ? e14.message : e14}`
